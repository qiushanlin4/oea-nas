cd /etc/pam.d
rm login
write work/login.fixed login
sif login mode 0100644
sif login uid 0
sif login gid 0
cd /etc
rm securetty
write work/securetty.new securetty
sif securetty mode 0100644
sif securetty uid 0
sif securetty gid 0
cd /etc/systemd/system
rm netup.service
write work/netup.service netup.service
sif netup.service mode 0100644
cd /etc/NetworkManager/system-connections
rm eth0.nmconnection
write work/eth0.nmconnection eth0.nmconnection
sif eth0.nmconnection mode 0100600
cd /etc/systemd/system/serial-getty@.service.d
rm override.conf
write work/serial-override.conf override.conf
sif override.conf mode 0100644
